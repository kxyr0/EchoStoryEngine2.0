cmake_minimum_required(VERSION 3.24)
project(EchoNpcStudio VERSION 0.15.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "ECHO NPC STUDIO requires Windows and MSVC with the Windows SDK")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/shaders/Minecraft.hlsl" SKIN_SHADER)
configure_file(src/graphics/Shader.h.in generated/Shader.h @ONLY)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/locales/ru.json" RUSSIAN_JSON)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/locales/en.json" ENGLISH_JSON)
configure_file(src/localization/Defaults.h.in generated/LocalizationDefaults.h @ONLY)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/templates/background.json" BACKGROUND_TEMPLATE_JSON)
configure_file(src/npc/TemplateDefaults.h.in generated/TemplateDefaults.h @ONLY)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS locales/ru.json locales/en.json templates/background.json)
add_library(echo_core STATIC src/skin/Skin.cpp src/npc/Repository.cpp src/npc/Templates.cpp src/npc/GameExport.cpp src/npc/Removal.cpp src/graphics/MinecraftModel.cpp src/portrait/FaceRig.cpp src/portrait/FaceAuthoring.cpp src/localization/Localization.cpp)
target_include_directories(echo_core PUBLIC src)
target_include_directories(echo_core PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_compile_definitions(echo_core PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(echo_core PUBLIC /utf-8 /W4 /permissive- /EHsc)
target_link_libraries(echo_core PUBLIC windowscodecs ole32)
add_executable(EchoNpcStudio WIN32 src/main.cpp src/app/Application.cpp src/graphics/Graphics.cpp src/ui/UI.cpp src/ui/Panels.cpp src/ui/PortraitPanel.cpp src/ui/FaceWizard.cpp src/ui/NameColorPicker.cpp tests/StudioSmoke.cpp)
target_link_options(EchoNpcStudio PRIVATE "/MANIFEST:EMBED" "/MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/src/app/app.manifest")
target_include_directories(EchoNpcStudio PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_link_libraries(EchoNpcStudio PRIVATE echo_core d3d11 dxgi d3dcompiler d2d1 dwrite shell32 comdlg32 imm32 comctl32)
include(CTest)
if(BUILD_TESTING)
  add_executable(EchoCoreTests tests/CoreTests.cpp src/ui/UI.cpp src/graphics/Graphics.cpp)
  target_include_directories(EchoCoreTests PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
  target_link_libraries(EchoCoreTests PRIVATE echo_core d3d11 dxgi d3dcompiler d2d1 dwrite imm32)
  add_test(NAME EchoCoreTests COMMAND EchoCoreTests)
  add_executable(EchoLocalizationTests tests/LocalizationTests.cpp)
  target_link_libraries(EchoLocalizationTests PRIVATE echo_core)
  add_test(NAME EchoLocalizationTests COMMAND EchoLocalizationTests "${CMAKE_CURRENT_SOURCE_DIR}")
endif()
install(TARGETS EchoNpcStudio RUNTIME DESTINATION .)
install(DIRECTORY locales DESTINATION .)
install(DIRECTORY templates DESTINATION .)
add_custom_command(TARGET EchoNpcStudio POST_BUILD
  COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/locales" "$<TARGET_FILE_DIR:EchoNpcStudio>/locales"
  COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/templates" "$<TARGET_FILE_DIR:EchoNpcStudio>/templates")
